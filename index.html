C Language Data Structure: Singly Linked List Implementation

Written by Ashish Kaushik • Complete memory-safe dynamic allocation implementation in standard C.

#include <stdio.h>
#include <stdlib.h>

// Node Definition for Dynamic Linked List
typedef struct Node {
    int data;
    struct Node* next;
} Node;

// Create a new node with malloc
Node* create_node(int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    if (!newNode) {
        printf("Memory Allocation Failed!\n");
        return NULL;
    }
    newNode->data = data;
    newNode->next = NULL;
    return newNode;
}

// Insert at beginning
void insert_head(Node** head, int data) {
    Node* newNode = create_node(data);
    newNode->next = *head;
    *head = newNode;
}

// Print entire list
void print_list(Node* head) {
    Node* curr = head;
    printf("List: ");
    while (curr != NULL) {
        printf("[%d] -> ", curr->data);
        curr = curr->next;
    }
    printf("NULL\n");
}

// Free all memory to avoid leaks
void free_list(Node* head) {
    Node* temp;
    while (head != NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }
}
